4. Scroll calculation fix — clamp lightY so it reaches the bottom of the bar exactly when the bottom of the section hits the center of the viewport, not the bottom of the page:
tsxconst sectionEl = sectionRef.current;
const barEl = barRef.current;
if (!sectionEl || !barEl) return;

const sectionRect = sectionEl.getBoundingClientRect();
const barHeight = barEl.offsetHeight;
const viewportCenter = window.innerHeight * 0.5;

// Progress: 0 when section top hits viewport center, 1 when section bottom hits viewport center
const progress = Math.max(0, Math.min(1,
  (viewportCenter - sectionRect.top) / sectionRect.height
));

const lightY = progress * (barHeight - 120); // 120 = light height
setLightY(lightY);
